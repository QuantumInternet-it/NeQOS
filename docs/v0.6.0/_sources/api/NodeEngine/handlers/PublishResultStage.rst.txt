Behavior
--------

Assemble and validate the output set, write the result file in the current run
directory, then expose the exports. Shared and scratch references are resolved
at handler execution. A post-publication failure retains evidence for operators;
WaitMicroProtocolStage and child_output do not import a failed child's result.

State and results
-----------------

Writes schemaVersion 0.1, templateProtocolId, runtimeProtocolId and outputs.
Exposes the validated outputs and records PublishResult in taskResults.
Missing failIfScratchPath values do not trigger failure. Summary generation
and the optional failure check occur after the result file is written.

Failure conditions
------------------

Invalid mappings/references, absent source values, conflicting paths or
output-schema mismatch fail. Invalid fileName, unavailable run directory,
result/summary write errors or a collision with summary.md fail. A truthy
writeSummary is not enough: its value must be a boolean. Post-publication
failure can leave result.json and exports available to an operator.

          """Validate, write and publish this protocol's declared result.
      
          Assemble and validate the output set, write the result file in the current run
          directory, then expose the exports. Shared and scratch references are resolved
          at handler execution. A post-publication failure retains evidence for operators;
          WaitMicroProtocolStage and child_output do not import a failed child's result.
      
          Parameters:
              exports (array; required): Non-empty objects with from (scratchpad path) and as (declared outputSchema export path).
              fileName (string; default: "result.json"): Plain relative filename in this run's artifact directory.
              writeSummary (boolean; default: false): Also write summary.md with the published outputs and any post-publication failure reason.
              failIfScratchPath (string; default: null): Scratchpad path checked after publication; a present value other than null/empty string becomes the failure reason.
      
          State and results:
              Writes schemaVersion 0.1, templateProtocolId, runtimeProtocolId and outputs.
              Exposes the validated outputs and records PublishResult in taskResults.
              Missing failIfScratchPath values do not trigger failure. Summary generation
              and the optional failure check occur after the result file is written.
      
          Failure conditions:
              Invalid mappings/references, absent source values, conflicting paths or
              output-schema mismatch fail. Invalid fileName, unavailable run directory,
              result/summary write errors or a collision with summary.md fail. A truthy
              writeSummary is not enough: its value must be a boolean. Post-publication
              failure can leave result.json and exports available to an operator.
          """
          try:
              resolvedParams = _ResolveSharedReferences(params, ctx)
              exports = _NormalizeMappingEntries(resolvedParams, "exports")
          except ValueError as exc:
              return StageResult(decision="fail", reason=str(exc))
          if not exports:
              return StageResult(decision="fail", reason="PublishResultStage requires at least one export entry.")
          fileName = str(resolvedParams.get("fileName") or "result.json").strip()
          writeSummary = resolvedParams.get("writeSummary", False)
          if not isinstance(writeSummary, bool):
              return StageResult(decision="fail", reason="writeSummary must be a boolean.")
          if writeSummary and fileName == "summary.md":
              return StageResult(decision="fail", reason="Result filename must not collide with summary.md.")
          filePath = Path(fileName)
          if not fileName or filePath.is_absolute() or len(filePath.parts) != 1 or filePath.name != fileName:
              return StageResult(decision="fail", reason="fileName must be a plain relative filename.")
          pi = ctx.nodeEngine.microProtocolInstances.get(ctx.protocolId)
          if pi is None:
              return StageResult(decision="fail", reason=f"Cannot publish result for unknown protocolId={ctx.protocolId!r}.")
          protocolExports = dict(pi.metadata.get("exports") or {})
          try:
              for entry in exports:
                  sourcePath = _NormalizePath(entry["from"], fieldName="from")
                  targetPath = _NormalizePath(entry["as"], fieldName="as")
                  value = _GetPath(ctx.scratchpad, sourcePath)
                  if value is _MISSING:
                      return StageResult(
                          decision="fail",
                          reason=f"Scratchpad path {entry['from']!r} is not available for result publication.",
                      )
                  _SetPath(protocolExports, targetPath, deepcopy(value))
              ctx.nodeEngine.ValidateProtocolExports(pi, protocolExports)
          except ValueError as exc:
              return StageResult(decision="fail", reason=f"Result contract validation failed: {exc}")
          runDirectory = str(pi.metadata.get("artifactRunDirectory") or "").strip()
          if not runDirectory:
              return StageResult(decision="fail", reason="Protocol run directory is unavailable for result publication.")
          artifactPath = Path(runDirectory) / fileName
          payload = {
              "schemaVersion": "0.1",
              "templateProtocolId": str(pi.templateProtocolId or pi.contract.protocolId),
              "runtimeProtocolId": ctx.protocolId,
              "outputs": protocolExports,
          }
          try:
              artifactPath.parent.mkdir(parents=True, exist_ok=True)
              artifactPath.write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8")
          except (OSError, TypeError, ValueError) as exc:
              return StageResult(decision="fail", reason=f"Could not write result artifact: {exc}")
          pi.metadata["exports"] = protocolExports
          publication = {
              "status": "OK",
              "taskName": "PublishResult",
              "result": {"artifactPath": str(artifactPath), "exports": exports},
          }
          ctx.taskResults.append(publication)
          publishedFailureReason = None
          failurePath = resolvedParams.get("failIfScratchPath")
          if failurePath not in (None, ""):
              try:
                  normalizedFailurePath = _NormalizePath(str(failurePath), fieldName="failIfScratchPath")
                  failureReason = _GetPath(ctx.scratchpad, normalizedFailurePath)
              except ValueError as exc:
                  return StageResult(decision="fail", reason=str(exc))
              if failureReason is not _MISSING and failureReason not in (None, ""):
                  publishedFailureReason = str(failureReason)
          if writeSummary:
              summaryPath = Path(runDirectory) / "summary.md"
              try:
                  summaryPath.write_text(RenderProtocolResultSummary(
                      payload, resultFileName=fileName, failureReason=publishedFailureReason,
                      exportPaths=(list(pi.contract.outputSchema["exports"])
                                   if pi.contract.outputSchema is not None
                                   else [entry["as"] for entry in exports]),
                  ), encoding="utf-8")
              except (OSError, TypeError, ValueError) as exc:
                  reason = f"Could not write result summary: {exc}"
                  if publishedFailureReason:
                      reason = f"{publishedFailureReason} Additionally, {reason}"
                  return StageResult(decision="fail", reason=reason)
              publication["result"]["summaryPath"] = str(summaryPath)
          if publishedFailureReason:
              return StageResult(decision="fail", reason=publishedFailureReason)
          return StageResult(decision="advance")
