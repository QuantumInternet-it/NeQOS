WaitMicroProtocolStage
======================

Wait for a child runtime submitted under a protocol key.

Behavior
--------

The handler reads the runtime ID saved by ``SubmitMicroProtocolStage``. It
requeues while the child is active, advances when it succeeds, and fails
when it reaches a failed or aborted state. On success, optional ``imports``
copy declared child exports into the parent scratchpad. ``pollDelaySec``
controls the requeue delay and defaults to 2 seconds.

      async def WaitMicroProtocolStage(ctx: StageContext, params: Dict[str, Any]) -> StageResult:
          """Wait for a child runtime submitted under a protocol key.
      
          The handler reads the runtime ID saved by ``SubmitMicroProtocolStage``. It
          requeues while the child is active, advances when it succeeds, and fails
          when it reaches a failed or aborted state. On success, optional ``imports``
          copy declared child exports into the parent scratchpad. ``pollDelaySec``
          controls the requeue delay and defaults to 2 seconds.
          """
      
          try:
              resolvedParams = _ResolveSharedReferences(params, ctx)
          except ValueError as exc:
              return StageResult(decision="fail", reason=str(exc))
          templateId = str(resolvedParams.get("templateId") or "")
          protocolKey = str(resolvedParams.get("protocolKey") or templateId)
          pollDelaySec = float(resolvedParams.get("pollDelaySec", 2.0))
          runtimeId = ctx.get(f"_mp_{protocolKey}")
          if runtimeId is None:
              return StageResult(
                  decision="fail",
                  reason=f"No runtime ID for protocolKey {protocolKey!r} — was SubmitMicroProtocolStage called first?",
              )
          if ctx.microprotocol_succeeded(runtimeId):
              try:
                  imports = _NormalizeMappingEntries(resolvedParams, "imports")
              except ValueError as exc:
                  return StageResult(decision="fail", reason=str(exc))
              if imports:
                  child = ctx.nodeEngine.microProtocolInstances.get(runtimeId)
                  if child is None:
                      return StageResult(
                          decision="fail",
                          reason=f"Cannot import from unknown microprotocol runtimeId={runtimeId!r}.",
                      )
                  childExports = dict(child.metadata.get("exports") or {})
                  imported: List[Dict[str, str]] = []
                  try:
                      for entry in imports:
                          sourcePath = _NormalizePath(entry["from"], fieldName="from")
                          targetPath = _NormalizePath(entry["as"], fieldName="as")
                          sourceName = ".".join(sourcePath)
                          if not ctx.nodeEngine.IsChildOutputPathDeclared(child, sourceName):
                              return StageResult(
                                  decision="fail",
                                  reason=(
                                      f"Child microprotocol {runtimeId!r} does not declare "
                                      f"output path {sourceName!r}."
                                  ),
                              )
                          value = _GetPath(childExports, sourcePath)
                          if value is _MISSING:
                              return StageResult(
                                  decision="fail",
                                  reason=(
                                      f"Child microprotocol {runtimeId!r} did not export "
                                      f"path {entry['from']!r}."
                                  ),
                              )
                          _SetPath(ctx.scratchpad, targetPath, deepcopy(value))
                          imported.append({"from": entry["from"], "as": entry["as"]})
                  except ValueError as exc:
                      return StageResult(decision="fail", reason=str(exc))
                  ctx.taskResults.append({
                      "status": "OK",
                      "taskName": "ImportMicroProtocolExports",
                      "result": {
                          "runtimeId": runtimeId,
                          "imports": imported,
                      },
                  })
              return StageResult(decision="advance")
          if ctx.microprotocol_failed(runtimeId):
              state = ctx.microprotocol_state(runtimeId)
              child = ctx.nodeEngine.microProtocolInstances.get(runtimeId)
              childError = str(getattr(child, "lastError", None) or "").strip()
              reason = f"MicroProtocol {templateId!r} ended with state={state}."
              if childError:
                  reason += f" Child error: {childError}"
              return StageResult(decision="fail", reason=reason)
          return StageResult(decision="requeue", delaySec=pollDelaySec)
