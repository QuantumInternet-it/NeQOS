      async def RunSingleTaskStage(ctx: StageContext, params: Dict[str, Any]) -> StageResult:
          """Run one DeviceOS task and fail unless its reply is explicitly successful."""
          deviceId = str(params["deviceId"])
          taskName = str(params["taskName"])
          reply = await ctx.run_task(
              deviceId=deviceId,
              taskName=taskName,
              params=dict(params.get("taskParams") or {}),
          )
          resultKey = str(params.get("resultKey") or "").strip()
          if resultKey:
              ctx.set(resultKey, {
                  "deviceId": deviceId,
                  "taskName": taskName,
                  "reply": reply,
              })
          if bool(params.get("failOnTaskError", True)) and str(reply.get("status") or "").upper() != "OK":
              return StageResult(
                  decision="fail",
                  reason=(
                      f"{taskName} failed on {deviceId}: "
                      f"{reply.get('errorType')} {reply.get('errorMessage')}"
                  ),
              )
          return StageResult(decision=str(params.get("decision") or "advance"))
